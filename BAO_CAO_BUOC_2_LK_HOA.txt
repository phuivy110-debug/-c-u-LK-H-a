C. CẤU TRÚC PRODUCT CUỐI CÙNG (/src/types.ts)
--------------------------------------------------------------------------------
export type ProductStatus = 'active' | 'inactive';

export interface Product {
  id: string;
  name: string;
  title: string;              // Alias hỗ trợ mã cũ
  slug: string;               // URL Slug không dấu
  category: CategoryId | string;
  description?: string;
  imageUrl?: string;
  image: string;              // Alias hỗ trợ mã cũ
  price?: number;
  dealPrice?: number;         // Alias hỗ trợ mã cũ
  originalPrice?: number;
  discountPercent?: number;
  shopeeUrl?: string;
  affiliateUrl?: string;      // Alias hỗ trợ mã cũ
  tiktokUrl?: string;
  status: ProductStatus;      // 'active' | 'inactive'
  featured: boolean;          // true | false
  isFeatured?: boolean;       // Alias hỗ trợ mã cũ
  updatedAt?: string;

  // Metadata thực từ Google Sheet (Chỉ hiện khi có dữ liệu)
  shopName?: string;
  rating?: number;
  soldCount?: string;
  couponCode?: string;
  badges?: BadgeType[];
  isMall?: boolean;
}


D. BẢNG MAPPING CỘT GOOGLE SHEET SANG TRƯỜNG PRODUCT
--------------------------------------------------------------------------------
+ Cột Google Sheet            + Trường Product        + Quy tắc xử lý
--------------------------------------------------------------------------------
- Mã SP / ID                  : id                    : Giữ nguyên ID gốc. Nếu trống -> prod-[slug]
- Tên sản phẩm / Tên          : name / title          : Bắt buộc. Dùng để tạo slug chuẩn SEO.
- Hạng mục / Danh mục         : category              : Chuẩn hóa về: rods, reels, baits, lines, floats, accessories.
- Giá bán / Giá ưu đãi        : price / dealPrice     : Đưa về số dương. Trống hoặc sai format -> undefined.
- Giá gốc / Niêm yết          : originalPrice         : Đưa về số dương. Trống hoặc sai format -> undefined.
- Link Affiliate Shopee       : shopeeUrl             : Validate hostname Shopee. Giữ nguyên query parameters.
- Link Affiliate TikTok       : tiktokUrl             : Validate hostname TikTok. Giữ nguyên query parameters.
- Link ảnh / Ảnh sản phẩm     : imageUrl / image      : URL direct. Trống -> tự động extract từ Shopee link nếu có.
- Mô tả / Thông tin           : description           : Lấy nguyên văn. Trống -> undefined.
- Trạng thái                  : status                : 'inactive' nếu điền ẩn/inactive. Còn lại -> 'active'.
- Nổi bật                     : featured              : true nếu điền 1/có/true. Còn lại -> false.
